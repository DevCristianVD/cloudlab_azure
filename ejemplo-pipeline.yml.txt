# ARCHIVO DE REFERENCIA — NO ACTIVAR JUNTO AL WORKFLOW CREADO POR AZURE
# Si se usa el flujo manual (origen Other), guardar bajo .github/workflows/deploy.yml
# y configurar el secreto AZURE_STATIC_WEB_APPS_API_TOKEN en GitHub.
name: Desplegar CloudLab a Azure
on:
  push:
    branches: [main]
  workflow_dispatch:
permissions:
  contents: read
jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - name: Obtener código
        uses: actions/checkout@v4
      - name: Publicar en Azure Static Web Apps
        uses: Azure/static-web-apps-deploy@v1
        with:
          azure_static_web_apps_api_token: ${{ secrets.AZURE_STATIC_WEB_APPS_API_TOKEN }}
          action: upload
          app_location: /
          api_location: ''
          output_location: ''
